script
async function enviarTeste(scriptId, scriptName, btn) {
    const original = btn.innerHTML;
    btn.disabled = true;
    btn.innerHTML = 'span class=spinner-border spinner-border-smspan Enviando...';
    try {
        const r = await fetch(`testessend${scriptId}`, {method 'POST'});
        const data = await r.json();
        if (data.ok) {
            showToast('success', data.message  'Teste enviado ✓');
        } else {
            showToast('danger', data.error  'Erro');
        }
    } catch (e) {
        showToast('danger', String(e));
    } finally {
        btn.disabled = false;
        btn.innerHTML = original;
        setTimeout(() = location.reload(), 1500);
    }
}

function showToast(type, msg) {
    const box = document.getElementById('alertBox');
    const div = document.createElement('div');
    div.className = `alert alert-${type} shadow-sm`;
    div.style.minWidth = '280px';
    div.textContent = msg;
    box.appendChild(div);
    setTimeout(() = div.remove(), 4000);
}
script
